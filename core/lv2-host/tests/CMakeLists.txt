# A native LV2 bundle with metadata, audio/control/atom ports and deliberately bad variants.
set(bundle "${CMAKE_CURRENT_BINARY_DIR}/test fixture.lv2")
file(MAKE_DIRECTORY "${bundle}")
configure_file(manifest.ttl.in "${bundle}/manifest.ttl" @ONLY)
configure_file(fixture.ttl "${bundle}/fixture.ttl" COPYONLY)
add_library(oscairn-lv2-fixture MODULE fixture.cpp)
target_link_libraries(oscairn-lv2-fixture PRIVATE PkgConfig::LILV)
set_target_properties(oscairn-lv2-fixture PROPERTIES PREFIX "" OUTPUT_NAME fixture LIBRARY_OUTPUT_DIRECTORY "${bundle}")
add_executable(oscairn-lv2-test host.cpp)
target_include_directories(oscairn-lv2-test PRIVATE ../../clap-host/tests)
target_link_libraries(oscairn-lv2-test PRIVATE oscairn_control_json PkgConfig::LILV)
target_compile_options(oscairn-lv2-test PRIVATE -Wall -Wextra)
add_dependencies(oscairn-lv2-test oscairn-lv2-fixture oscairn_step_sequencer oscairn_basic_synth)
add_test(NAME lv2-host COMMAND oscairn-lv2-test "${bundle}" $<TARGET_FILE:oscairn_step_sequencer> $<TARGET_FILE:oscairn_basic_synth>)
set_tests_properties(lv2-host PROPERTIES TIMEOUT 30)
